115function originAllowed(req) {
116 const o = req.headers.origin;
117 if (!o) return true;
118 if (o === 'null') return config.cors.allowFile;
119 if (config.cors.origins.indexOf(o) !== -1) return true;
120 try {
121 const u = new URL(o);
122 if (config.cors.allowLocal &&
123 (u.hostname === '127.0.0.1' || u.hostname === 'localhost' || u.hostname === '::1')) return true;
124 const host = req.headers.host || '';
125 if (host && u.host === host) return true;
126 } catch (e) { /* ignore */ }
127 return false;
128}
