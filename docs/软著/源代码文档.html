104function staticDenied(rel) {
105 const segs = rel.split('/').filter(Boolean);
106 for (const seg of segs) {
107 if (seg.startsWith('.')) return true; // .env.json / .git / .workbuddy
108 if (DENY_DIRS.has(seg.toLowerCase())) return true; // 后端与文档目录
109 }
110 return !STATIC_EXT.has(path.extname(rel).toLowerCase()); // .db / .log / .md / .txt / .pdf
111}

113function serveStatic(req, res, pathname) {
114 let rel = decodeURIComponent(pathname);
115 if (rel === '/' || rel === '') rel = '/index.html';
116 if (staticDenied(rel)) {
117 res.writeHead(404, { 'Content-Type': 'text/plain; charset=utf-8' });
118 res.end('404 Not Found');
119 return;
120 }
121 // 防目录穿越
122 const target = path.normalize(path.join(config.root, rel));
123 if (!target.startsWith(config.root)) { res.writeHead(403); res.end('Forbidden'); return; }
124 fs.stat(target, (err, st) => {
125 if (err || !st.isFile()) {
126 res.writeHead(404, { 'Content-Type': 'text/plain; charset=utf-8' });
127 res.end('404 Not Found: ' + rel);
128 return;
129 }
130 const ext = path.extname(target).toLowerCase();
131 res.writeHead(200, {
132 'Content-Type': MIME[ext] || 'application/octet-stream',
133 'Content-Length': st.size
134 });
135 fs.createReadStream(target).pipe(res);
136 });
137}
